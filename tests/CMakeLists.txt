cmake_minimum_required(VERSION 3.22)
project(clock_host_tests LANGUAGES CXX)
enable_testing()

target_include_directories(test_core PRIVATE ../main)

target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -UNDEBUG)
